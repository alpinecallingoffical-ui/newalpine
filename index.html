Technical PRD & Architecture Blueprint

Production Architecture Blueprint for Alpine Digital Marketplace (Nepal localized)

1. PostgreSQL / Supabase Schema Schema Blueprint

-- 1. USERS & PROFILES
CREATE TABLE public.profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
  email TEXT NOT NULL,
  phone TEXT,
  full_name TEXT,
  role TEXT DEFAULT 'customer' CHECK (role IN ('customer', 'admin')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. CATEGORIES
CREATE TABLE public.categories (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  icon TEXT
);

-- 3. PRODUCTS (EBooks, Topups, Subscriptions, Audio, Apps)
CREATE TABLE public.products (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  title TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  category_id UUID REFERENCES public.categories(id),
  type TEXT CHECK (type IN ('file', 'key_voucher')),
  price_npr NUMERIC NOT NULL,
  description TEXT,
  cover_image TEXT,
  file_storage_path TEXT, -- Secured Supabase Bucket Path
  activation_pool TEXT[], -- Encrypted keys array for topups/subs
  stock_status BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. ORDERS & FULFILLMENT
CREATE TABLE public.orders (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  total_amount NUMERIC NOT NULL,
  payment_method TEXT CHECK (payment_method IN ('esewa', 'khalti', 'fonepay')),
  payment_status TEXT DEFAULT 'pending' CHECK (payment_status IN ('pending', 'completed', 'failed')),
  transaction_ref TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE public.order_items (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  order_id UUID REFERENCES public.orders(id) ON DELETE CASCADE,
  product_id UUID REFERENCES public.products(id),
  price_npr NUMERIC NOT NULL,
  delivered_key TEXT -- Assigned key/voucher if applicable
);

-- 5. VERIFIED REVIEWS
CREATE TABLE public.reviews (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id UUID REFERENCES public.products(id),
  user_id UUID REFERENCES public.profiles(id),
  rating INTEGER CHECK (rating BETWEEN 1 AND 5),
  comment TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

2. Supabase Row Level Security (RLS) Policy Blueprint

-- Enable RLS
ALTER TABLE public.products ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY;

-- Products: Everyone reads, Admins write
CREATE POLICY "Public Products Access" ON public.products FOR SELECT USING (true);

-- Orders: Users can only see their own orders
CREATE POLICY "Users view own orders" ON public.orders 
  FOR SELECT USING (auth.uid() = user_id);

-- Storage Bucket Security (Secured Downloads)
-- Signed URLs with expiration are generated via Edge Functions AFTER validating order completion
CREATE POLICY "Strict Digital Asset Access" ON storage.objects 
  FOR SELECT USING (
    bucket_id = 'digital-assets' 
    AND EXISTS (
      SELECT 1 FROM public.orders o
      JOIN public.order_items oi ON oi.order_id = o.id
      JOIN public.products p ON oi.product_id = p.id
      WHERE o.user_id = auth.uid() 
      AND o.payment_status = 'completed'
      AND p.file_storage_path = name
    )
  );
